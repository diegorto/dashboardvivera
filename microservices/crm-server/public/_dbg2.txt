<script>
let usersCache  EQ  [];
let dealSearchTimer  EQ  null;

function fmtTime(iso) {
  const d  EQ  new Date(iso);
  return d.toLocaleString('pt-BR', { day COLON  '2-digit', month COLON  '2-digit', hour COLON  '2-digit', minute COLON  '2-digit' });
}

function groupLabel(dueAt) {
  const now  EQ  new Date();
  const due  EQ  new Date(dueAt);
  const startOfToday  EQ  new Date(now.getFullYear(), now.getMonth(), now.getDate());
  const startOfTomorrow  EQ  new Date(startOfToday.getTime() + 86400000);
  const startOfNext7  EQ  new Date(startOfToday.getTime() + 7 * 86400000);
  if (due < startOfToday) return 'Atrasadas';
  if (due < startOfTomorrow) return 'Hoje';
  if (due < new Date(startOfTomorrow.getTime() + 86400000)) return 'Amanha';
  if (due < startOfNext7) return 'Esta semana';
  return 'Depois';
}

async function loadUsers() {
  const data  EQ  await api('/api/crm/ui/users');
  usersCache  EQ  data.users || [];
  const userFilter  EQ  document.getElementById('userFilter');
  const actUser  EQ  document.getElementById('actUser');
  usersCache.forEach(u  EQ > {
    const opt1  EQ  document.createElement('option'); opt1.value  EQ  u.id; opt1.textContent  EQ  u.name;
    userFilter.appendChild(opt1);
    const opt2  EQ  document.createElement('option'); opt2.value  EQ  u.id; opt2.textContent  EQ  u.name;
    actUser.appendChild(opt2);
  });
}

async function loadAgenda() {
  const container  EQ  document.getElementById('agendaContainer');
  const totalLabel  EQ  document.getElementById('totalLabel');
  totalLabel.textContent  EQ  'Carregando...';
  container.innerHTML  EQ  '';
  const userId  EQ  document.getElementById('userFilter').value;
  const includeDone  EQ  document.getElementById('includeDone').checked;
  try {
    let url  EQ  '/api/crm/ui/agenda?includeDone EQ ' + includeDone;
    if (userId) url + EQ  '&userId EQ ' + encodeURIComponent(userId);
    const data  EQ  await api(url);
    const activities  EQ  data.activities || [];
    totalLabel.textContent  EQ  activities.length + ' atividade(s)';
    if (!activities.length) {
      container.innerHTML  EQ  '<p style EQ "color COLON #888;padding COLON 16px 0;">Nenhuma atividade agendada.</p>';
      return;
    }
    const groups  EQ  {};
    const order  EQ  ['Atrasadas', 'Hoje', 'Amanha', 'Esta semana', 'Depois'];
    activities.forEach(a  EQ > {
      const g  EQ  groupLabel(a.due_at);
      if (!groups[g]) groups[g]  EQ  [];
      groups[g].push(a);
    });
    let html  EQ  '';
    order.forEach(g  EQ > {
      if (!groups[g] || !groups[g].length) return;
      html + EQ  `<div class EQ "agenda-group"><h3>${g}</h3>`;
      groups[g].forEach(a  EQ > {
        const overdue  EQ  g  EQ  EQ  EQ  'Atrasadas' ? 'overdue'  COLON  '';
        html + EQ  `
          <div class EQ "agenda-item ${overdue}" data-id EQ "${a.id}">
            <input type EQ "checkbox" class EQ "chk act-done-chk" data-id EQ "${a.id}" ${a.done ? 'checked'  COLON  ''}>
            <div class EQ "time">${fmtTime(a.due_at)}</div>
            <span class EQ "type-badge">${a.type || '-'}</span>
            <div class EQ "info" ${a.deal_id ? `onclick EQ "window.location.href EQ 'detail.html?id EQ ${a.deal_id}'" style EQ "cursor COLON pointer;"`  COLON  ''}>
              <div class EQ "subj">${a.subject || a.content || '(sem assunto)'}</div>
              <div class EQ "meta">${a.patient_name || a.deal_title || ''}${a.user_name ? ' - ' + a.user_name  COLON  ''}</div>
            </div>
          </div>
        `;
      });
      html + EQ  '</div>';
    });
    container.innerHTML  EQ  html;
    document.querySelectorAll('.act-done-chk').forEach(chk  EQ > {
      chk.addEventListener('click', async (e)  EQ > {
        e.stopPropagation();
        const id  EQ  chk.getAttribute('data-id');
        try {
          await api('/api/crm/activities/' + id, { method COLON  'PATCH', body COLON  JSON.stringify({ done COLON  chk.checked }) });
          loadAgenda();
        } catch (err) { showToast ? showToast(err.message, 'error')  COLON  alert(err.message); }
      });
    });
  } catch (e) {
    totalLabel.textContent  EQ  'Erro ao carregar COLON  ' + e.message;
  }
}

document.getElementById('dealSearchInput').addEventListener('input', (e)  EQ > {
  clearTimeout(dealSearchTimer);
  const q  EQ  e.target.value.trim();
  const box  EQ  document.getElementById('dealSuggestions');
  if (!q) { box.classList.remove('open'); return; }
  dealSearchTimer  EQ  setTimeout(async ()  EQ > {
    try {
      const data  EQ  await api('/api/crm/ui/deals?search EQ ' + encodeURIComponent(q) + '&pageSize EQ 8');
      const deals  EQ  data.deals || data.rows || [];
      if (!deals.length) { box.innerHTML  EQ  '<div style EQ "color COLON #888;">Nenhum resultado</div>'; box.classList.add('open'); return; }
      box.innerHTML  EQ  deals.map(d  EQ > `<div data-id EQ "${d.id}" data-label EQ "${(d.patient_name || d.title || '')} - ${d.title || ''}">${d.patient_name || '-'} - ${d.title || ''} (${d.patient_phone || ''})</div>`).join('');
      box.classList.add('open');
      box.querySelectorAll('div[data-id]').forEach(el  EQ > {
        el.addEventListener('click', ()  EQ > {
          document.getElementById('selectedDealId').value  EQ  el.getAttribute('data-id');
          document.getElementById('selectedDealLabel').textContent  EQ  'Selecionado COLON  ' + el.getAttribute('data-label');
          document.getElementById('dealSearchInput').value  EQ  el.getAttribute('data-label');
          box.classList.remove('open');
        });
      });
    } catch (err) { /* ignore */ }
  }, 300);
});

document.getElementById('saveActivityBtn').addEventListener('click', async ()  EQ > {
  const dealId  EQ  document.getElementById('selectedDealId').value;
  const subject  EQ  document.getElementById('actSubject').value.trim();
  const dueAtLocal  EQ  document.getElementById('actDueAt').value;
  if (!subject) { alert('Informe o assunto'); return; }
  if (!dueAtLocal) { alert('Informe data e hora'); return; }
  const dueAtUtc  EQ  new Date(dueAtLocal).toISOString().slice(0, 19).replace('T', ' ');
  try {
    await api('/api/crm/activities', {
      method COLON  'POST',
      body COLON  JSON.stringify({
        dealId COLON  dealId || null,
        type COLON  document.getElementById('actType').value,
        subject,
        content COLON  document.getElementById('actNote').value || subject,
        dueAt COLON  dueAtUtc,
        assignedUserId COLON  document.getElementById('actUser').value || null,
        duration COLON  document.getElementById('actDuration').value || null
      })
    });
    closeModal();
    loadAgenda();
  } catch (err) { alert(err.message); }
